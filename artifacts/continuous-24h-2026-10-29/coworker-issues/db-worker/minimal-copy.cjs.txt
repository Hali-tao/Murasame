const { cpSync, mkdirSync, mkdtempSync, readFileSync, writeFileSync } = require('node:fs')
const { join } = require('node:path')
const { tmpdir } = require('node:os')
const source = mkdtempSync(join(tmpdir(), 'murasame-db-copy-source-'))
const target = mkdtempSync(join(tmpdir(), 'murasame-db-copy-target-'))
mkdirSync(join(source, 'nested'))
writeFileSync(join(source, 'nested', 'fixture.txt'), 'only a synthetic copy fixture')
console.log(JSON.stringify({ node: process.version, source, target }))
console.log('before cpSync')
cpSync(source, target, { recursive: true })
console.log(`after cpSync roundtrip=${readFileSync(join(target, 'nested', 'fixture.txt'), 'utf8') === 'only a synthetic copy fixture'}`)
