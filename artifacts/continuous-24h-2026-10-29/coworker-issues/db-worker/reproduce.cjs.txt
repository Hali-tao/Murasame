const { appendFileSync, cpSync, mkdtempSync, existsSync, readFileSync, readdirSync, rmSync } = require('node:fs')
const { execFileSync } = require('node:child_process')
const { join } = require('node:path')
const { tmpdir } = require('node:os')
const root = process.cwd()
const marker = join(__dirname, `${process.argv[2]}.markers.txt`)
function step(message) { appendFileSync(marker, `${message}\n`); process.stdout.write(`${message}\n`) }
step(`node=${process.version} temp=${tmpdir()}`)
const fixture = mkdtempSync(join(tmpdir(), 'murasame-database-recovery-'))
step(`created primary ${fixture}`)
const env = { ...process.env, MURASAME_DATABASE_FIXTURE: fixture }
delete env.ELECTRON_RUN_AS_NODE
const entry = join(root, 'out/electron-tests/database-recovery.cjs')
const electron = require('electron')
step('before first execFileSync')
const output = execFileSync(electron, [entry, 'fresh-restart'], { cwd: root, encoding: 'utf8', timeout: 30000, env })
step(`after first execFileSync PASS=${output.includes('PASS database-recovery fresh-restart')} items=${readdirSync(fixture).length}`)
step('before restart execFileSync')
const restarted = execFileSync(electron, [entry, 'fresh-restart-verify'], { cwd: root, encoding: 'utf8', timeout: 30000, env })
step(`after restart execFileSync PASS=${restarted.includes('PASS database-recovery fresh-restart-verify')} LocalState=${existsSync(join(fixture, 'Local State'))}`)
const savedKey = readFileSync(join(fixture, 'murasame-db-key.json'))
const restored = mkdtempSync(join(tmpdir(), 'murasame-database-recovery-'))
step(`created restored ${restored}`)
step('before cpSync')
cpSync(fixture, restored, { recursive: true })
step(`after cpSync items=${readdirSync(restored).length}`)
step('before restored execFileSync')
const reopened = execFileSync(electron, [entry, 'fresh-restart-verify'], { cwd: root, encoding: 'utf8', timeout: 30000, env: { ...env, MURASAME_DATABASE_FIXTURE: restored } })
step(`after restored execFileSync PASS=${reopened.includes('PASS database-recovery fresh-restart-verify')} sameKey=${savedKey.equals(readFileSync(join(restored, 'murasame-db-key.json')))}`)
step('diagnostic complete; generated profiles retained for inspection')
